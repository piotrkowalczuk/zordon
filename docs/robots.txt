# Cloudflare PREPENDS its managed content-signals block above everything here
# (it blocks AI crawlers; Googlebot is not among them). That leaves two
# `User-agent: *` groups in the served file — Google merges same-agent groups
# per RFC 9309, so the rules below do apply.
User-agent: *

# The llms.txt corpus reproduces every page's content as one plain-text
# document, which cannot carry a canonical tag. Left crawlable it competes
# with the pages it mirrors. Agents fetch it directly via the
# `link: rel="service-doc"` header, which robots.txt does not affect.
Disallow: /llms.txt
Disallow: /llms-full.txt
